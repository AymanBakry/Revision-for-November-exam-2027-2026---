const KEY="1523"; function doPost(e){ const d=JSON.parse(e.postData.contents); SpreadsheetApp.getActiveSheet().appendRow([d.time,d.seat,d.name,d.score,d.total,d.percent,d.mistakes,d.report]); return ContentService.createTextOutput("ok"); } function doGet(e){ const out=o=>ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); if(e.parameter.key!==KEY)return out({error:"denied"}); const rows=SpreadsheetApp.getActiveSheet().getDataRange().getValues().map(r=>({time:String(r[0]),seat:String(r[1]),name:r[2],score:+r[3],total:+r[4],percent:+r[5],mistakes:String(r[6]),report:String(r[7])})); return out(rows); }
